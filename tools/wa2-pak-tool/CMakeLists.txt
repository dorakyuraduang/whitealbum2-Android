cmake_minimum_required(VERSION 3.20)
project(WA2PakTool LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This desktop tool currently targets Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(pak_core STATIC pak.cpp encoding_windows.cpp integrity_windows.cpp)
target_include_directories(pak_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(pak_core PUBLIC UNICODE _UNICODE)
target_link_libraries(pak_core PUBLIC bcrypt)
add_library(pak_i18n STATIC i18n.cpp)
target_link_libraries(pak_i18n PUBLIC pak_core)
add_executable(WA2_PAK_Tool WIN32 main_win32.cpp)
target_link_libraries(WA2_PAK_Tool PRIVATE pak_i18n comctl32 ole32 shell32 uuid advapi32)
add_executable(wa2_pak_tests tests.cpp ../../engine/resource_archive.cpp)
target_include_directories(wa2_pak_tests PRIVATE ../..)
target_link_libraries(wa2_pak_tests PRIVATE pak_i18n)
if(MSVC)
    foreach(target pak_core pak_i18n WA2_PAK_Tool wa2_pak_tests)
        target_compile_options(${target} PRIVATE /W4 /utf-8)
    endforeach()
else()
    foreach(target pak_core pak_i18n WA2_PAK_Tool wa2_pak_tests)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endforeach()
    target_link_options(WA2_PAK_Tool PRIVATE -municode "-Wl,--subsystem,windows" -static -s)
    target_link_options(wa2_pak_tests PRIVATE -static)
endif()
