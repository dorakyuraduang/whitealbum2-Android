cmake_minimum_required(VERSION 3.20)
project(WA2SharedHostRefresh LANGUAGES C CXX)
add_executable(wa2_prepare_lzss "${CMAKE_CURRENT_LIST_DIR}/../../scripts/pack_lzss.cpp")
get_filename_component(WA2_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
# Exercise the shipped, unmetered static bodies. Historical comparison builds
# require deleted baseline/oracle objects and are not silently substituted.
set(WA2_SHARED_AOT_CHECKS OFF CACHE BOOL "" FORCE)
set(WA2_AOT_DIAGNOSTICS OFF CACHE BOOL "" FORCE)
set(CMAKE_CXX_STANDARD 20)
add_subdirectory("${WA2_ROOT}/recompiled" shared_aot)
foreach(name selected_save_files portable_system resource_mount test_profile portable_scheduler function_process subtitles portable_shell fast_forward script_wait_callback save_inventory profile_batch io_diagnostics resource_reuse resource_loading loose_resources pc_loose_resources novel_cpu text_control sprite_texture_refs host_progress image_chroma language_resources language_names language_prompts ukrainian external_subtitles music_flag_saves subtitle_spacing)
    add_executable(wa2_refresh_${name} "${WA2_ROOT}/tests/${name}.cpp")
    target_link_libraries(wa2_refresh_${name} PRIVATE wa2_function_process)
endforeach()
add_executable(wa2_refresh_movie_codecs "${WA2_ROOT}/tests/movie_codecs.cpp")
target_link_libraries(wa2_refresh_movie_codecs PRIVATE wa2_function_process wa2_movie)
add_executable(wa2_refresh_movie_formats "${WA2_ROOT}/tests/movie_formats.cpp")
target_link_libraries(wa2_refresh_movie_formats PRIVATE wa2_function_process wa2_movie)
add_executable(wa2_refresh_movie_bridge "${WA2_ROOT}/tests/portable_movie_bridge.cpp")
target_link_libraries(wa2_refresh_movie_bridge PRIVATE wa2_function_process wa2_movie)
target_link_libraries(wa2_refresh_image_chroma PRIVATE wa2_chroma_reference)
if(WIN32)
    target_sources(wa2_refresh_subtitle_spacing PRIVATE "${WA2_ROOT}/platform/subtitle_font_windows.cpp")
    target_link_libraries(wa2_refresh_subtitle_spacing PRIVATE gdi32)
endif()
