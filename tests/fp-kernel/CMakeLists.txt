cmake_minimum_required(VERSION 3.20)
project(WA2PortableFloatingPoint LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
get_filename_component(WA2_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_library(wa2_extended STATIC "${WA2_ROOT}/recompiled/extended80.cpp" "${WA2_ROOT}/recompiled/transcend.cpp")
target_include_directories(wa2_extended PUBLIC "${WA2_ROOT}/recompiled" "${WA2_ROOT}/tools/WindowsRecompTool-main/include")
set_target_properties(wa2_extended PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    set_source_files_properties("${WA2_ROOT}/recompiled/extended80.cpp" PROPERTIES COMPILE_OPTIONS "-O2")
endif()
add_library(wa2_fp_forms STATIC "${WA2_ROOT}/build/fp-kernel-input/cases.cpp" "${WA2_ROOT}/build/fp-kernel-input/simd_cases.cpp")
target_include_directories(wa2_fp_forms PUBLIC "${WA2_ROOT}/tests")
target_link_libraries(wa2_fp_forms PUBLIC wa2_extended)
if(ANDROID)
    set_target_properties(wa2_fp_forms PROPERTIES POSITION_INDEPENDENT_CODE ON)
    target_compile_definitions(wa2_fp_forms PRIVATE WA2_FP_HOST_ORACLE=0)
    add_library(wa2_fp_link_check SHARED entry.cpp)
    target_link_libraries(wa2_fp_link_check PRIVATE "-Wl,--whole-archive" wa2_fp_forms wa2_extended "-Wl,--no-whole-archive")
    target_link_options(wa2_fp_link_check PRIVATE "-Wl,--no-undefined" "-Wl,-z,max-page-size=16384")
else()
    target_compile_definitions(wa2_fp_forms PRIVATE WA2_FP_HOST_ORACLE=1)
    add_library(wa2_fp_native STATIC IMPORTED)
    set_target_properties(wa2_fp_native PROPERTIES IMPORTED_LOCATION "${WA2_ROOT}/build/wa2-win64/winrecomp/libwinrecomp_native.a")
    add_executable(wa2_fp_tests "${WA2_ROOT}/tests/portable_fp.cpp")
    target_link_libraries(wa2_fp_tests PRIVATE wa2_fp_forms wa2_fp_native)
    add_executable(wa2_simd_tests "${WA2_ROOT}/tests/portable_simd.cpp")
    target_link_libraries(wa2_simd_tests PRIVATE wa2_fp_forms wa2_fp_native)
    add_executable(wa2_fp_execution_tests "${WA2_ROOT}/tests/fp_execution.cpp"
        "${WA2_ROOT}/tools/WindowsRecompTool-main/src/runtime/native_memory.cpp")
    target_link_libraries(wa2_fp_execution_tests PRIVATE wa2_extended wa2_fp_native)
    add_executable(wa2_fp_workload_tests "${WA2_ROOT}/tests/fp_workload.cpp")
    target_link_libraries(wa2_fp_workload_tests PRIVATE wa2_extended wa2_fp_native)
    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
        target_compile_options(wa2_fp_workload_tests PRIVATE -Oz)
    endif()
endif()
