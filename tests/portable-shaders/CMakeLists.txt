cmake_minimum_required(VERSION 3.20)
project(WA2PortableShaders LANGUAGES C CXX)
include("${CMAKE_CURRENT_LIST_DIR}/../../platform/render/shaders.cmake")
if(NOT ANDROID)
    add_executable(wa2_portable_shader_tests ../portable_shaders.cpp)
    target_link_libraries(wa2_portable_shader_tests PRIVATE wa2_fixed_pipeline wa2_common_shader_assembler wa2_common_shader_gl)
else()
    add_library(wa2_portable_shader_link_check SHARED entry.cpp)
    target_link_libraries(wa2_portable_shader_link_check PRIVATE "-Wl,--whole-archive" wa2_fixed_pipeline wa2_common_shader_assembler wa2_common_shader_parser wa2_common_shader_gl "-Wl,--no-whole-archive")
    target_link_options(wa2_portable_shader_link_check PRIVATE "-Wl,--no-undefined" "-Wl,-z,max-page-size=16384")
endif()
