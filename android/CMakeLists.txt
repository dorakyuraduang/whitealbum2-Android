cmake_minimum_required(VERSION 3.20)
project(WA2Android LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(wa2_frame_metrics STATIC native/frame_metrics.cpp)
target_include_directories(wa2_frame_metrics PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/native")
set_target_properties(wa2_frame_metrics PROPERTIES POSITION_INDEPENDENT_CODE ON)

if(NOT ANDROID)
    add_executable(wa2_android_frame_metrics_tests ../tests/android_frame_metrics.cpp)
    target_link_libraries(wa2_android_frame_metrics_tests PRIVATE wa2_frame_metrics)
    enable_testing()
    add_test(NAME android_frame_metrics COMMAND wa2_android_frame_metrics_tests "${CMAKE_BINARY_DIR}/frame-metrics-tests.json")
    return()
endif()
if(NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "This port targets Android ARM64 only")
endif()
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_OPENGLES ON CACHE BOOL "" FORCE)
set(SDL_OPENGL OFF CACHE BOOL "" FORCE)
set(SDL_VULKAN OFF CACHE BOOL "" FORCE)
# SDLActivity creates HIDDeviceManager even before the native game starts.
# Its Java native declarations require the Android HID JNI implementation.
set(SDL_HIDAPI ON CACHE BOOL "" FORCE)
set(SDL_SENSOR OFF CACHE BOOL "" FORCE)
set(SDL_RENDER ON CACHE BOOL "" FORCE)
set(SDL_RENDER_D3D OFF CACHE BOOL "" FORCE)
set(SDL_DIRECTX OFF CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../tools/SDL-release-2.32.10" sdl2 EXCLUDE_FROM_ALL)
target_link_options(SDL2 PRIVATE "-Wl,-z,max-page-size=16384")
# The platform entry compiles the same static game functions as Windows.
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../hosts" shared_host)
