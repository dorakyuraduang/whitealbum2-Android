cmake_minimum_required(VERSION 3.20)
project(WA2SharedHosts LANGUAGES C CXX)
get_filename_component(WA2_ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
if(NOT TARGET wa2_function_process)
    set(WA2_SHARED_AOT_CHECKS OFF CACHE BOOL "" FORCE)
    set(WA2_AOT_DIAGNOSTICS OFF CACHE BOOL "" FORCE)
    add_subdirectory("${WA2_ROOT}/recompiled" shared_runtime)
endif()
add_library(wa2_shared_startup STATIC shared/startup.cpp)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${WA2_ROOT}/config/release.json")
file(READ "${WA2_ROOT}/config/release.json" WA2_RELEASE_JSON)
string(JSON WA2_HOST_VERSION GET "${WA2_RELEASE_JSON}" name)
target_compile_definitions(wa2_shared_startup PRIVATE "WA2_HOST_VERSION=\"${WA2_HOST_VERSION}\"")
target_compile_features(wa2_shared_startup PUBLIC cxx_std_20)
target_link_libraries(wa2_shared_startup PUBLIC wa2_function_process)
set_target_properties(wa2_shared_startup PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(ANDROID)
    if(NOT ANDROID_ABI STREQUAL "arm64-v8a")
        message(FATAL_ERROR "The shared Android host requires ARM64")
    endif()
    add_library(wa2_android SHARED android_main.cpp
        "${WA2_ROOT}/android/native/android_resources.cpp"
        "${WA2_ROOT}/android/native/compiled_resources.cpp"
        "${WA2_ROOT}/android/native/frame_metrics.cpp"
        "${WA2_ROOT}/android/native/frame_metrics_jni.cpp")
    target_include_directories(wa2_android PRIVATE "${WA2_ROOT}/android/native" "${WA2_ROOT}/engine" "${WA2_ROOT}/platform")
    target_link_libraries(wa2_android PRIVATE wa2_shared_startup log android)
    target_link_options(wa2_android PRIVATE "-Wl,--no-undefined" "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL" "-Wl,-Bsymbolic")
else()
    add_executable(wa2_native WIN32 windows_main.cpp)
    target_link_libraries(wa2_native PRIVATE wa2_shared_startup)
    set_target_properties(wa2_native PROPERTIES OUTPUT_NAME WA2_SDL2_x64)
endif()
