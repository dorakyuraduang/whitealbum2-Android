cmake_minimum_required(VERSION 3.20)
project(WA2NativeAndroid LANGUAGES C CXX)
get_filename_component(WA2_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "The native Android host currently supports ARM64 only")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_OPENGLES ON CACHE BOOL "" FORCE)
set(SDL_OPENGL OFF CACHE BOOL "" FORCE)
set(SDL_VULKAN OFF CACHE BOOL "" FORCE)
# SDLActivity's HIDDeviceManager registers JNI callbacks before SDL_main.
# Its real implementation must be present even without a connected controller.
set(SDL_HIDAPI ON CACHE BOOL "" FORCE)
set(SDL_SENSOR OFF CACHE BOOL "" FORCE)
set(SDL_RENDER ON CACHE BOOL "" FORCE)
set(SDL_RENDER_D3D OFF CACHE BOOL "" FORCE)
set(SDL_DIRECTX OFF CACHE BOOL "" FORCE)
add_subdirectory("${WA2_ROOT}/tools/SDL-release-2.32.10" sdl2 EXCLUDE_FROM_ALL)
target_link_options(SDL2 PRIVATE "-Wl,-z,max-page-size=16384")
find_package(Threads REQUIRED)
add_subdirectory("${WA2_ROOT}/engine" engine EXCLUDE_FROM_ALL)
target_link_libraries(wa2_engine PUBLIC Threads::Threads)
include("${WA2_ROOT}/engine/movie.cmake")
add_subdirectory("${WA2_ROOT}/game" game EXCLUDE_FROM_ALL)
include("${WA2_ROOT}/platform/native_services.cmake")
include("${WA2_ROOT}/android/game/entry.cmake")
wa2_add_native_android_entry(wa2_native_android)
# This source-only migration root does not select a released APK. Native Java
# adapters must use android_source_sets.java_sources('native'); no legacy
# widget/shell transport or generated game function belongs in its package.
