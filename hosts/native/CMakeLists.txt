cmake_minimum_required(VERSION 3.20)
project(WA2NativeDesktop LANGUAGES C CXX)
get_filename_component(WA2_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Only the Windows x64 desktop codec/toolchain is configured; game sources are shared with Android")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_HIDAPI OFF CACHE BOOL "" FORCE)
set(SDL_JOYSTICK OFF CACHE BOOL "" FORCE)
set(SDL_HAPTIC OFF CACHE BOOL "" FORCE)
set(SDL_SENSOR OFF CACHE BOOL "" FORCE)
set(SDL_DIRECTX OFF CACHE BOOL "" FORCE)
set(SDL_RENDER_D3D OFF CACHE BOOL "" FORCE)
set(SDL_OPENGL ON CACHE BOOL "" FORCE)
set(SDL_VULKAN OFF CACHE BOOL "" FORCE)
add_subdirectory("${WA2_ROOT}/tools/SDL-release-2.32.10" sdl2 EXCLUDE_FROM_ALL)
find_package(Threads REQUIRED)
add_subdirectory("${WA2_ROOT}/engine" engine EXCLUDE_FROM_ALL)
target_link_libraries(wa2_engine PUBLIC Threads::Threads)
include("${WA2_ROOT}/engine/movie.cmake")
add_subdirectory("${WA2_ROOT}/game" game EXCLUDE_FROM_ALL)
include("${WA2_ROOT}/platform/native_services.cmake")
include("${CMAKE_CURRENT_LIST_DIR}/entry.cmake")
wa2_add_native_desktop_entry(wa2_native_desktop)
set_target_properties(wa2_native_desktop PROPERTIES OUTPUT_NAME WA2_Native_x64)
# This entry is incomplete and is not a replacement for the retained AOT
# release. Never package its component/link checks as a complete game port.
